package com.example.hiltpractice

import android.app.Application
import android.content.Context
import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModelProvider
import androidx.room.Room
import com.example.hiltpractice.data.*
import com.example.hiltpractice.ui.NotesViewModel
import kotlinx.coroutines.Dispatchers
import okhttp3.OkHttpClient
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory

class AppContainer(private val context: Context) {

    val okHttpClient: OkHttpClient by lazy {
        println("created OkHttpClient")
        OkHttpClient.Builder().build()
    }

    val retrofit: Retrofit by lazy {
        println("created Retrofit")
        Retrofit.Builder()
            .baseUrl("https://api.github.com/")
            .client(okHttpClient)
            .addConverterFactory(GsonConverterFactory.create())
            .build()
    }

    val gitHubService: GitHubService by lazy { retrofit.create(GitHubService::class.java) }

    val database: AppDatabase by lazy {
        println("created AppDatabase")
        Room.databaseBuilder(context, AppDatabase::class.java, "app.db").build()
    }

    fun noteRepository(): NoteRepository {
        println("created NoteRepository")
        return RoomNoteRepository(database.noteDao(), Dispatchers.IO)
    }

    fun userRepository(): UserRepository {
        println("created UserRepository")
        return UserRepositoryImpl(gitHubService, database.userDao(), Dispatchers.IO)
    }
}

class ManualApp : Application() {
    val container by lazy { AppContainer(this) }
}

class NotesViewModelFactory(private val container: AppContainer) : ViewModelProvider.Factory {
    @Suppress("UNCHECKED_CAST")
    override fun <T : ViewModel> create(modelClass: Class<T>): T =
        NotesViewModel(container.noteRepository()) as T
}
